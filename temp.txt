    // 4️⃣ Prevent invalid group state
    if (group.members.length <= 2) {
        throw new Error("Group must have at least 2 members")
    }

    // 5️⃣ Remove member
    await prisma.fYPGroupMember.delete({
        where: {
            id: targetMembership.id,
        },
    })

    return {
        removedStudentId: targetStudentId,
        groupId: group.id,
    }
}
